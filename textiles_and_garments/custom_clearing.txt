    document.head.appendChild(script);
    
    // Load reCAPTCHA
    var recaptchaScript = document.createElement('script');
    recaptchaScript.src = 'https://www.google.com/recaptcha/api.js';
    document.head.appendChild(recaptchaScript);
    
    // Load Google Maps with your API key
    if (typeof google == 'undefined') {
      var mapsScript = document.createElement('script');
      mapsScript.src = 'https://maps.googleapis.com/maps/api/js?key=AIzaSyDtgyfEuGyhaV7mQb9sWuX50DbAds5YEQo&libraries=places&callback=initAutocomplete';
      document.head.appendChild(mapsScript);
    }
  } else {
    initCustomsForm();
  }
});

function initCustomsForm() {
  // TOGGLE SECTIONS
  document.querySelectorAll('.section-toggle').forEach(toggle => {
    toggle.addEventListener('click', function() {
      const targetId = this.getAttribute('data-target');
      const targetSection = document.getElementById(targetId);
      if (targetSection) {
        targetSection.classList.toggle('hidden');
        // Update the toggle text
        this.textContent = this.textContent.includes('+') ? 
          this.textContent.replace('+', '-') : 
          this.textContent.replace('-', '+');
      }
    });
  });

  // ADD CARGO ROW
  if (document.getElementById('addCargoBtn')) {
    document.getElementById('addCargoBtn').addEventListener('click', function() {
      const tbody = document.querySelector('#cargoTable tbody');
      if (tbody) {
        const newRow = document.createElement('tr');
        newRow.innerHTML = `
          <td><input name="marks[]"></td>
          <td><input name="packages[]"></td>
          <td><input name="description[]"></td>
          <td><input name="gross_mass[]"></td>
          <td><input name="cubic[]"></td>
          <td><input name="tariff[]"></td>
          <td><input name="value[]"></td>
          <td><input name="supplier[]"></td>
        `;
        tbody.appendChild(newRow);
      }
    });
  }

  // SIGNATURE PAD
  const canvas = document.getElementById('signature-pad');
  if (canvas) {
    const ctx = canvas.getContext('2d');
    let drawing = false;

    canvas.addEventListener('mousedown', () => drawing = true);
    canvas.addEventListener('mouseup', () => drawing = false);
    canvas.addEventListener('mousemove', draw);

    function draw(e) {
      if (!drawing) return;
      ctx.lineWidth = 2;
      ctx.lineCap = 'round';
      ctx.strokeStyle = '#000';
      ctx.lineTo(e.offsetX, e.offsetY);
      ctx.stroke();
      ctx.beginPath();
      ctx.moveTo(e.offsetX, e.offsetY);
    }

    if (document.getElementById('clearSignatureBtn')) {
      document.getElementById('clearSignatureBtn').addEventListener('click', function() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
      });
    }
  }

  // TRANSPORT MODE TOGGLE
  document.querySelectorAll('input[name="transport_mode"]').forEach(radio => {
    radio.addEventListener('change', function() {
      document.querySelectorAll('.transport-section').forEach(section => {
        section.classList.add('hidden');
      });
      const targetSection = document.getElementById(this.value + 'Fields');
      if (targetSection) {
        targetSection.classList.remove('hidden');
      }
    });
  });

  // SAME AS CLIENT FUNCTIONALITY WITH CODE AUTO-FILL
  if (document.getElementById('sameAsClientExporter')) {
    document.getElementById('sameAsClientExporter').addEventListener('change', function() {
      const clientFields = ['client_name', 'client_vat', 'client_email', 'contact_number', 'client_address', 'client_ie_code'];
      const exporterFields = ['exporter_name', 'exporter_vat', 'exporter_email', 'exporter_phone', 'exporter_address', 'exporter_code'];
      
      if (this.checked) {
        clientFields.forEach((field, i) => {
          const clientValue = document.querySelector(`input[name="${field}"]`)?.value;
          const exporterInput = document.querySelector(`input[name="${exporterFields[i]}"]`);
          if (exporterInput && clientValue !== undefined) {
            exporterInput.value = clientValue;
          }
        });
      } else {
        exporterFields.forEach(field => {
          const exporterInput = document.querySelector(`input[name="${field}"]`);
          if (exporterInput) {
            exporterInput.value = '';
          }
        });
      }
    });
  }

  if (document.getElementById('sameAsClientImporter')) {
    document.getElementById('sameAsClientImporter').addEventListener('change', function() {
      const clientFields = ['client_name', 'client_vat', 'client_email', 'contact_number', 'client_address', 'client_ie_code'];
      const importerFields = ['importer_name', 'importer_vat', 'importer_email', 'importer_phone', 'importer_address', 'importer_code'];
      
      if (this.checked) {
        clientFields.forEach((field, i) => {
          const clientValue = document.querySelector(`input[name="${field}"]`)?.value;
          const importerInput = document.querySelector(`input[name="${importerFields[i]}"]`);
          if (importerInput && clientValue !== undefined) {
            importerInput.value = clientValue;
          }
        });
      } else {
        importerFields.forEach(field => {
          const importerInput = document.querySelector(`input[name="${field}"]`);
          if (importerInput) {
            importerInput.value = '';
          }
        });
      }
    });
  }

  // TEMPORARY IMPORT MODAL
  const modal = document.getElementById('temporaryImportModal');
  if (modal) {
    const span = document.getElementsByClassName('close')[0];

    // Add event listener to all temporary import checkboxes
    document.querySelectorAll('.temporary-import').forEach(checkbox => {
      checkbox.addEventListener('change', function() {
        if (this.checked) {
          modal.style.display = 'block';
        } else {
          // Clear any saved temporary import details if unchecked
          document.querySelector('input[name="temporary_import_details"]')?.remove();
        }
      });
    });

    // Close modal when clicking X
    span.onclick = function() {
      modal.style.display = 'none';
      document.querySelector('.temporary-import').checked = false;
    }

    // Close modal when clicking outside
    window.onclick = function(event) {
      if (event.target == modal) {
        modal.style.display = 'none';
        document.querySelector('.temporary-import').checked = false;
      }
    }

    // Calculate expiry date when duration changes
    const repairDuration = document.getElementById('repairDuration');
    if (repairDuration) {
      repairDuration.addEventListener('change', function() {
        const importDate = new Date(document.getElementById('tempImportDate').value);
        if (!isNaN(importDate.getTime())) {
          const expiryDate = new Date(importDate);
          expiryDate.setMonth(expiryDate.getMonth() + parseInt(this.value || 0));
          document.getElementById('expiryDate').value = expiryDate.toISOString().split('T')[0];
        }
      });
    }

    // Also calculate when date changes
    const tempImportDate = document.getElementById('tempImportDate');
    if (tempImportDate) {
      tempImportDate.addEventListener('change', function() {
        const duration = document.getElementById('repairDuration').value;
        if (duration) {
          const importDate = new Date(this.value);
          if (!isNaN(importDate.getTime())) {
            const expiryDate = new Date(importDate);
            expiryDate.setMonth(expiryDate.getMonth() + parseInt(duration));
            document.getElementById('expiryDate').value = expiryDate.toISOString().split('T')[0];
          }
        }
      });
    }

    // Save temporary import details
    const saveTempImport = document.getElementById('saveTempImport');
    if (saveTempImport) {
      saveTempImport.addEventListener('click', function() {
        const fields = ['tempImportDate', 'repairDuration', 'goodsValue', 'originCountry', 'goodsDescription'];
        let isValid = true;
        
        fields.forEach(field => {
          const element = document.getElementById(field);
          if (element) {
            if (!element.value) {
              isValid = false;
              element.style.borderColor = 'red';
            } else {
              element.style.borderColor = '#ddd';
            }
          }
        });
        
        if (isValid) {
          const details = {
            importDate: document.getElementById('tempImportDate').value,
            duration: document.getElementById('repairDuration').value,
            expiryDate: document.getElementById('expiryDate').value,
            value: document.getElementById('goodsValue').value,
            origin: document.getElementById('originCountry').value,
            description: document.getElementById('goodsDescription').value
          };
          
          // Create or update hidden input with the details
          let hiddenInput = document.querySelector('input[name="temporary_import_details"]');
          if (!hiddenInput) {
            hiddenInput = document.createElement('input');
            hiddenInput.type = 'hidden';
            hiddenInput.name = 'temporary_import_details';
            document.getElementById('customsForm').appendChild(hiddenInput);
          }
          hiddenInput.value = JSON.stringify(details);
          
          modal.style.display = 'none';
        } else {
          alert('Please fill in all temporary import details');
        }
      });
    }
  }

  // DOWNLOAD FORM
  if (document.getElementById('downloadBtn')) {
    document.getElementById('downloadBtn').addEventListener('click', function() {
      const formData = new FormData(document.getElementById('customsForm'));
      const data = {};
      formData.forEach((value, key) => {
        data[key] = value;
      });
      
      // In a real implementation, you would send this to your server to generate a PDF
      alert('In a real implementation, this would download a PDF with the form data');
    });
  }

  // WHATSAPP SHARE
  if (document.getElementById('whatsappBtn')) {
    document.getElementById('whatsappBtn').addEventListener('click', function() {
      const clientName = document.querySelector('input[name="client_name"]')?.value || '';
      const message = `Customs Clearing Form Submission\nClient: ${clientName}\n\nPlease review the full details.`;
      window.open(`https://wa.me/27662078745?text=${encodeURIComponent(message)}`, '_blank');
    });
  }

  // FORM SUBMISSION
  if (document.getElementById('customsForm')) {
    document.getElementById('customsForm').addEventListener('submit', function(e) {
      e.preventDefault();
      
      // Verify reCAPTCHA
      if (typeof grecaptcha !== 'undefined' && !grecaptcha.getResponse()) {
        alert('Please complete the reCAPTCHA verification');
        return;
      }
      
      // Check honeypot field
      const honeypot = document.getElementById('mmcargo-security');
      if (honeypot && honeypot.value) {
        alert('Invalid submission');
        return;
      }
      
      // Validate required fields
      let isValid = true;
      document.querySelectorAll('[required]').forEach(field => {
        if (!field.value) {
          field.style.borderColor = 'red';
          isValid = false;
        } else {
          field.style.borderColor = '#ddd';
        }
      });
      
      if (!isValid) {
        alert('Please fill in all required fields');
        return;
      }
      
      // Check signature
      if (canvas && canvas.toDataURL() === canvas.toDataURL('image/png', 0)) {
        alert('Please provide your signature');
        return;
      }
      
      // Prepare form data
      const formData = new FormData(this);
      if (canvas) {
        formData.append('signature', canvas.toDataURL());
      }
      
      // In a real implementation, you would send this to your server
      // Here we'll just show the collected data
      const formObject = {};
      formData.forEach((value, key) => {
        // Handle array fields
        if (key.endsWith('[]')) {
          const baseKey = key.slice(0, -2);
          if (!formObject[baseKey]) {
            formObject[baseKey] = [];
          }
          formObject[baseKey].push(value);
        } else {
          // Try to parse JSON for temporary import details
          if (key === 'temporary_import_details' && value) {
            try {
              formObject[key] = JSON.parse(value);
            } catch (e) {
              formObject[key] = value;
            }
          } else {
            formObject[key] = value;
          }
        }
      });
      
      console.log('Form submission data:', formObject);
      alert('Form submitted successfully!\n\nCheck console for complete form data.');
      
      // Reset form
      this.reset();
      if (canvas) {
        const ctx = canvas.getContext('2d');
        ctx.clearRect(0, 0, canvas.width, canvas.height);
      }
      if (typeof grecaptcha !== 'undefined') grecaptcha.reset();
    });
  }
}

// GOOGLE PLACES AUTOCOMPLETE
function initAutocomplete() {
  const addressFields = ['client_address', 'exporter_address', 'importer_address'];
  
  addressFields.forEach(fieldId => {
    const input = document.getElementById(fieldId);
    if (input && typeof google !== 'undefined') {
      const autocomplete = new google.maps.places.Autocomplete(input, {
        types: ['geocode']
      });
      
      autocomplete.addListener('place_changed', function() {
        const place = autocomplete.getPlace();
        if (!place.geometry) {
          console.log("No details available for input: '" + place.name + "'");
          return;
        }
      });
    }
  });
}
